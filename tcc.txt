CADASTRO DE MEDICOS :

CREATE TABLE medicos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150),
    crm VARCHAR(20),
    especialidade VARCHAR(100),
    email VARCHAR(150),
    senha VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

PACIENTES:
CREATE TABLE pacientes(

    id INT PRIMARY KEY AUTO_INCREMENT,

    nome VARCHAR(150),

    foto VARCHAR(255),

    genero VARCHAR(30),

    tipo_sanguineo VARCHAR(5),

    alergias TEXT,

    peso VARCHAR(20),

    altura VARCHAR(20)

);

CRISES:
CREATE TABLE crises(

    id INT PRIMARY KEY AUTO_INCREMENT,

    paciente_id INT,

    data_crise DATE,

    duracao VARCHAR(30),

    gravidade VARCHAR(30),

    descricao TEXT,

    FOREIGN KEY (paciente_id)
    REFERENCES pacientes(id)

);
